核心代码

.grid {
    display: grid;
    grid-template-areas:
        'a b b'
        'c d d'
        'c d e'
        'c d f';
    grid-template-columns: 1.2fr 1fr 1fr;
    grid-template-rows: 90px 90px 90px 90px;
    gap: 14px;
}
.area-a { grid-area: a; }
.area-b { grid-area: b; }
/* ... 其余区域同理 */
